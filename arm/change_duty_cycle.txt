void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
  HAL_UART_Receive_IT(&huart2, rxData, 1);
  if(rxData[0] == 'a')
  {
    TIM3->CCR1 = 300;
    rxData[0] = 0;
  }
  else if(rxData[0] == 'b')
  {
    TIM3->CCR1 = 500;
    rxData[0] = 0;
  }
  else if(rxData[0] == 'c')
  {
    TIM3->CCR1 = 700;
    rxData[0] = 0;
  }
  else if(rxData[0] == 'd')
  {
    TIM3->CCR1 = 0;
    rxData[0] = 0;
  }
}
