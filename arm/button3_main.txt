void ledRightShift(uint8_t num)
{
  for(uint8_t i = num; i > 0; i--)
  {
    HAL_GPIO_WritePin(led[i-1].port, led[i-1].number, led[i-1].onState);
    HAL_Delay(100);
  }
  HAL_Delay(500);
  for(uint8_t i = num; i > 0; i--)
  {
    HAL_GPIO_WritePin(led[i-1].port, led[i-1].number, led[i-1].offState);
    HAL_Delay(100);
  }
  HAL_Delay(500);
}
